# Oracle과 MySQL 비교


## 📌 MySQL과 문법 차이

##### 1. 자동 증가 컬럼 (Auto Increment)

```mysql
// MySQL
CREATE TABLE users (
	id INT AUTO_INCREMENT PRIMARY KEY,
	name VARCHAR(100)	
);
```
```sql
// Oracle (12c 이상)
CREATE TABLE users (
	id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	name VARCHAR2(100)
);
```


##### 2. 문자열 연결

```mysql
// MySQL
// 결과 : Hello World
SELECT CONCAT('Hello', ' ', 'World');
```
```sql
// Oracle
// 결과 : Hello World
SELECT 'Hello' || ' ' || 'World' FROM dual;
```


#### 3. NULL 처리 함수

```mysql
// MySQL (IFNULL, COALESCE)
SELECT IFNULL(phone, '없음') AS phone FROM users;
SELECT COALESCE(phone, '없음') AS phone FROM users;
```
```sql
// Oracle (NVL, NVL2)
SELECT NVL(phone, '없음') AS phone FROM users;
SELECT NVL2(phone, '있음', '없음') AS phone FROM users;
```


##### 4. 조건 분기 함수 (DECODE vs CASE)

```mysql
// MySQL (CASE)
SELECT
	CASE status
		WHEN 'A' THEN 'Active'
		WHEN 'I' THEN 'Inactive'
		ELSE 'Unknown'
	END AS status_desc
FROM users;
```
```sql
// Oracle (DECODE)
SELECT DECODE(status, 'A', 'Active', 'I', 'Inactive', 'Unknown') AS statis_desc
FROM users;
``` 


##### 5. 날짜 차이 계산

```mysql
// MySQL
// 결과: 9
SELECT DATEDIFF('2025-04-10', '2025-04-01') AS days_diff;
```
```sql
// Oracle
SELECT TO_DATE('2025-04-10', 'YYYY-MM-DD') - TO_DATE('2025-04-01', 'YYYY-MM-DD') FROM dual;
```


##### 6. 페이징 (상위 N개 행 조회)
```mysql
// MySQL
SELECT * FROM users LIMIT 5*
```
```sql
// Oracle(12c 이상)
SELECT * FROM users FETCH FIRST 5 ROWS ONLY;
```


##### 7. 계층형 쿼리 (조직 트리)
```mysql
// MySQL (WITH RECURSIVE)
WITH RECURSIVE emp_tree AS (
	SELECT employee_id, name, manager_id, 1 AS level
	FROM employees
	WHERE manager_id IS NULL
	UNION ALL
	SELECT e.employee_id, e.name, e.manager_id, et.level + 1
	FROM employees e
	JOIN emp_tree et ON e.manager_id = et.employee_id
)
```
```sql
// Oracle (CONNECT BY)
SELECT employee_id, name, manager_id
FROM employees
START WITH manager_id IS NULL
CONNECT BY PRIOR employee_id = manager_id;
```


##### 8. MERGE (UPSERT)
```mysql
// MySQL
// emp_id가 UNIQUE 또는 PRIMARY KEY일 때만 upsert 동작함
INSERT INTO employees (emp_id, name, salary)
VALUES (1, '홍길동', 5000)
ON DUPLICATE KEY UPDATE
	salary = VALUES(salary);
```
``` sql
// Oracle
MERGE INTO employees e
USING new_employees n
ON (e.emp_id = m.emp_id)
WHEN MATCHED THEN
	UPDATE SET e.salary = n.salary
WHEN NOT MATCHED THEN
	INSERT (emp_id, name, salary)
	VALUES (n.emp_id, n.name, n.salary);
```
